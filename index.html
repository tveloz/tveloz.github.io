#!/usr/bin/env node /** * Photo Manager Script for Academic Website * Helps manage photo carousel content and generates code for index.html */ const fs = require('fs'); const path = require('path'); class PhotoManager { constructor() { this.photosConfigPath = 'data/photos.json'; this.picsDirectory = 'pics/'; this.indexHtmlPath = 'index.html'; } // Load photos configuration loadConfig() { try { if (fs.existsSync(this.photosConfigPath)) { const config = JSON.parse(fs.readFileSync(this.photosConfigPath, 'utf8')); return config; } else { console.log('📁 No photos.json found, creating default configuration...'); return this.createDefaultConfig(); } } catch (error) { console.error('❌ Error loading photos config:', error); return this.createDefaultConfig(); } } // Create default configuration createDefaultConfig() { const defaultConfig = { carousel_settings: { rotation_speed: 8000, transition_effect: "fade", auto_start: true, random_start: true, pause_on_hover: true }, photos: [ { id: "conference_2024", filename: "AGI-2024.png", title: "Conference Participation", description: "Artificial General Intelligence Summit", location: "Panama City, Panama", date: "2024-01-15", tags: ["conference", "presentation", "research"], fallback_color: "#667eea", fallback_text: "Conference+Presentation", priority: 1 }, { id: "vub_lab_2024", filename: "Sole_talk_UVA_Joe.jpg", title: "Collaborating from VUB", description: "Talk by Richard Solé at UVA (Amsterdam) with Joe from UPD Philippines - SYMP lab collaborationst Centre Leo Apostel", location: "UVA Amsterdam, The Nethrelands", date: "2024-01-10", tags: ["research", "lab", "VUB", "SYMP"], fallback_color: "#764ba2", fallback_text: "Research+Lab", priority: 2 }, { id: "face_2024", filename: "face2024.jpg", title: "About me", description: "", location: "", date: "2024-01-10", tags: [""], fallback_color: "#f093fb", fallback_text: "About me", priority: 3 }, { id: "collaboration_2024", filename: "Japan_2023.jpg", title: "International Collaboration", description: "Research collaboration meeting with international partners", location: "Tokyo Japan", date: "2024-08-01", tags: ["collaboration", "international", "research"], fallback_color: "#4ade80", fallback_text: "Collaboration", priority: 4 } ], metadata: { last_updated: new Date().toISOString().split('T')[0], version: "1.0", total_photos: 4 } }; this.saveConfig(defaultConfig); return defaultConfig; } // Save configuration saveConfig(config) { try { // Ensure data directory exists const dataDir = path.dirname(this.photosConfigPath); if (!fs.existsSync(dataDir)) { fs.mkdirSync(dataDir, { recursive: true }); } // Update metadata config.metadata.last_updated = new Date().toISOString().split('T')[0]; config.metadata.total_photos = config.photos.length; fs.writeFileSync(this.photosConfigPath, JSON.stringify(config, null, 2)); console.log('✅ Photos configuration saved to', this.photosConfigPath); } catch (error) { console.error('❌ Error saving config:', error); } } // Check which photos exist checkPhotos() { const config = this.loadConfig(); const results = { existing: [], missing: [], total: config.photos.length }; console.log('🔍 Checking photo files...\n'); config.photos.forEach(photo => { const photoPath = path.join(this.picsDirectory, photo.filename); if (fs.existsSync(photoPath)) { results.existing.push(photo); console.log(`✅ ${photo.filename} - ${photo.title}`); } else { results.missing.push(photo); console.log(`❌ ${photo.filename} - ${photo.title} (MISSING)`); } }); console.log(`\n📊 Summary: ${results.existing.length}/${results.total} photos found`); if (results.missing.length > 0) { console.log('\n💡 Missing photos will use fallback placeholders'); console.log(' Upload photos to the pics/ directory with exact filenames'); } return results; } // Generate JavaScript code for HTML generateCarouselCode() { const config = this.loadConfig(); // Convert photos to the format expected by the HTML const photoData = config.photos.map(photo => ({ filename: photo.filename, description: photo.description, location: photo.location, fallback: `https://via.placeholder.com/400x400/${photo.fallback_color.replace('#', '')}/white?text=${photo.fallback_text}` })); const jsCode = ` // Photo carousel data - Auto-generated by photo-manager.js // Last updated: ${config.metadata.last_updated} const photoData = ${JSON.stringify(photoData, null, 12)}; // Carousel settings const carouselSettings = ${JSON.stringify(config.carousel_settings, null, 12)};`; return jsCode; } // Update index.html with new photo data updateIndexHtml() { try { if (!fs.existsSync(this.indexHtmlPath)) { console.error('❌ index.html not found'); return false; } let html = fs.readFileSync(this.indexHtmlPath, 'utf8'); const newCode = this.generateCarouselCode(); // Find and replace the photoData array const photoDataRegex = /\/\/ Photo carousel data[\s\S]*?const photoData = \[[\s\S]*?\];/; if (photoDataRegex.test(html)) { html = html.replace(photoDataRegex, newCode.trim()); fs.writeFileSync(this.indexHtmlPath, html); console.log('✅ index.html updated with new photo data'); return true; } else { console.log('⚠️ Could not find photoData section in index.html'); console.log('📝 Generated code (copy manually):'); console.log('====================================='); console.log(newCode); console.log('====================================='); return false; } } catch (error) { console.error('❌ Error updating index.html:', error); return false; } } // Add a new photo addPhoto(photoData) { const config = this.loadConfig(); // Add new photo const newPhoto = { id: photoData.id || `photo_${Date.now()}`, filename: photoData.filename, title: photoData.title || 'New Photo', description: photoData.description || 'Photo description', location: photoData.location || 'Location', date: photoData.date || new Date().toISOString().split('T')[0], tags: photoData.tags || ['academic'], fallback_color: photoData.fallback_color || '#667eea', fallback_text: photoData.fallback_text || 'New+Photo', priority: config.photos.length + 1 }; config.photos.push(newPhoto); this.saveConfig(config); console.log('✅ New photo added:', newPhoto.title); return newPhoto; } // Remove a photo removePhoto(photoId) { const config = this.loadConfig(); const initialLength = config.photos.length; config.photos = config.photos.filter(photo => photo.id !== photoId); if (config.photos.length < initialLength) { this.saveConfig(config); console.log('✅ Photo removed:', photoId); return true; } else { console.log('❌ Photo not found:', photoId); return false; } } // List all photos listPhotos() { const config = this.loadConfig(); console.log('📸 Current Photos:\n'); config.photos.forEach((photo, index) => { const exists = fs.existsSync(path.join(this.picsDirectory, photo.filename)); const status = exists ? '✅' : '❌'; console.log(`${index + 1}. ${status} ${photo.title}`); console.log(` File: ${photo.filename}`); console.log(` Description: ${photo.description}`); console.log(` Location: ${photo.location}`); console.log(` Date: ${photo.date}`); console.log(''); }); } // Setup directory structure setupDirectories() { const directories = ['pics', 'data']; directories.forEach(dir => { if (!fs.existsSync(dir)) { fs.mkdirSync(dir, { recursive: true }); console.log(`📁 Created directory: ${dir}/`); } }); } // Main help menu showHelp() { console.log(` 🎯 Photo Manager for Academic Website Usage: node photo-manager.js [command] Commands: check - Check which photos exist list - List all configured photos update - Update index.html with current photo data generate - Generate JavaScript code for manual copy setup - Create required directories help - Show this help Examples: node photo-manager.js check node photo-manager.js update node photo-manager.js list Configuration file: ${this.photosConfigPath} Photos directory: ${this.picsDirectory} `); } } // Command line interface function main() { const manager = new PhotoManager(); const command = process.argv[2] || 'help'; switch (command) { case 'check': manager.checkPhotos(); break; case 'list': manager.listPhotos(); break; case 'update': manager.setupDirectories(); manager.updateIndexHtml(); break; case 'generate': console.log('📝 Generated JavaScript code:'); console.log('====================================='); console.log(manager.generateCarouselCode()); console.log('====================================='); break; case 'setup': manager.setupDirectories(); manager.loadConfig(); // This creates default config console.log('✅ Setup complete'); break; case 'help': default: manager.showHelp(); break; } } // Export for use as module module.exports = PhotoManager; // Run if called directly if (require.main === module) { main(); }
